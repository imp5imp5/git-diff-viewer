cmake_minimum_required(VERSION 3.15)
project(GitDiffViewer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
include(CTest)
add_library(diff_core src/diff/UnifiedDiffParser.cpp src/diff/PresentationBuilder.cpp src/diff/ReviewComments.cpp)
target_include_directories(diff_core PUBLIC src)
if(WIN32)
  enable_language(RC)
  add_library(git_backend src/git/GitClient.cpp src/git/GitRepository.cpp)
  target_link_libraries(git_backend PUBLIC diff_core)
  add_executable(GitDiffViewer WIN32 src/main.cpp src/app/RepositoryController.cpp src/ui/MainWindow.cpp src/ui/DiffView.cpp src/ui/CommentEditor.cpp src/ui/CommitPicker.cpp src/ui/Settings.cpp src/ui/Screenshot.cpp src/ui/Automation.cpp resources/app.rc resources/app.manifest)
  set_target_properties(GitDiffViewer PROPERTIES OUTPUT_NAME gfd)
  target_link_libraries(GitDiffViewer PRIVATE git_backend user32 gdi32 shell32 ole32 comctl32 advapi32 windowscodecs uxtheme dwmapi)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/CoreTests.cpp)
  target_link_libraries(core_tests PRIVATE diff_core)
  add_test(NAME core COMMAND core_tests)
  if(WIN32)
    add_executable(settings_tests tests/SettingsTests.cpp src/ui/Settings.cpp)
    target_link_libraries(settings_tests PRIVATE diff_core shell32 advapi32)
    add_test(NAME settings COMMAND settings_tests)
    add_executable(comment_editor_tests tests/CommentEditorTests.cpp src/ui/CommentEditor.cpp)
    target_include_directories(comment_editor_tests PRIVATE src)
    target_link_libraries(comment_editor_tests PRIVATE user32 gdi32 comctl32 uxtheme dwmapi)
    add_test(NAME comment_editor COMMAND comment_editor_tests)
    add_executable(commit_picker_tests tests/CommitPickerTests.cpp src/ui/CommitPicker.cpp)
    target_include_directories(commit_picker_tests PRIVATE src)
    target_link_libraries(commit_picker_tests PRIVATE user32 gdi32 comctl32 uxtheme dwmapi)
    add_test(NAME commit_picker COMMAND commit_picker_tests)
    add_executable(git_tests tests/GitTests.cpp src/app/RepositoryController.cpp)
    target_link_libraries(git_tests PRIVATE git_backend)
    add_test(NAME git_integration COMMAND git_tests)
    set_tests_properties(git_integration PROPERTIES TIMEOUT 90)
    add_test(NAME interface COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
      -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/InterfaceTests.ps1"
      -Executable "$<TARGET_FILE:GitDiffViewer>"
      -Artifacts "${CMAKE_CURRENT_BINARY_DIR}/interface-$<CONFIG>")
    set_tests_properties(interface PROPERTIES TIMEOUT 90)
  endif()
endif()
